function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var action = data.action || "add"; if (action === "delete") { // 删除：在对应sheet里找到ID匹配的行删掉 var sheetName = data.mode === "pokemon" ? "pokemon流水" : "claude流水"; var sheet = ss.getSheetByName(sheetName); var rows = sheet.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { if (rows[i][rows[i].length - 1] === data.id) { sheet.deleteRow(i + 1); break; } } return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON); } if (action === "edit") { var sheetName = data.mode === "pokemon" ? "pokemon流水" : "claude流水"; var sheet = ss.getSheetByName(sheetName); var rows = sheet.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { if (rows[i][rows[i].length - 1] === data.id) { var date = new Date(data.timestamp); var dateStr = Utilities.formatDate(date, "Asia/Tokyo", "yyyy/MM/dd HH:mm"); if (data.mode === "pokemon") { var cost = data.cost || 0; var sell = data.sell || 0; sheet.getRange(i+1, 1, 1, 7).setValues([[dateStr, data.item, -cost, sell, sell-cost, data.note||"", data.id]]); } else { sheet.getRange(i+1, 1, 1, 6).setValues([[dateStr, data.item, data.category, -data.amount, data.note||"", data.id]]); } break; } } return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON); } // action === "add" var date = new Date(data.timestamp); var dateStr = Utilities.formatDate(date, "Asia/Tokyo", "yyyy/MM/dd HH:mm"); if (data.mode === "pokemon") { var sheet = ss.getSheetByName("pokemon流水"); var cost = data.cost || 0; var sell = data.sell || 0; sheet.appendRow([dateStr, data.item, -cost, sell, sell-cost, data.note||"", data.id]); } else { var sheet = ss.getSheetByName("claude流水"); sheet.appendRow([dateStr, data.item, data.category, -data.amount, data.note||"", data.id]); } return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({status:"error",message:err.toString()})).setMimeType(ContentService.MimeType.JSON); } }
